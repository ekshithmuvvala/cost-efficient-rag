# --- Web Framework & Interface ---
fastapi>=0.100.0          # Lets us expose /ingest and /query as HTTP endpoints
uvicorn>=0.22.0            # The server that actually runs our FastAPI app
pydantic>=2.0               # Used for config loading and request/response validation
pydantic-settings>=2.0    # Lets Pydantic read settings from a .env file automatically

# --- LLM Integration ---

# --- Evaluation & Metrics ---
pandas>=2.0.0        # Used to build and view result tables
numpy>=1.24.0        # Used for numeric metric calculations (nDCG, etc.)

# --- Utilities & Config ---
python-dotenv>=1.0.0   # Loads secrets/config out of a .env file
loguru>=0.7.0                # A friendlier version of Python's built-in logging
